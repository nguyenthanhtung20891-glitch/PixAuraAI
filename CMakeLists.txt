cmake_minimum_required(VERSION 3.22)
project(PixAuraCore VERSION 0.1.0 LANGUAGES C CXX)
option(PIXAURA_SANITIZERS "Enable host address/undefined behavior sanitizers" OFF)
if(PIXAURA_SANITIZERS AND (MSVC OR ANDROID))
    message(FATAL_ERROR "Combined ASan/UBSan validation requires a supported Clang/GCC host toolchain")
endif()

add_library(pixaura_core SHARED packages/core/src/core.cpp
    packages/core/src/document.cpp packages/core/src/document_api.cpp)
target_include_directories(pixaura_core PUBLIC packages/core/include)
target_compile_features(pixaura_core PUBLIC cxx_std_17)
target_compile_definitions(pixaura_core PUBLIC PIXAURA_SHARED PRIVATE PIXAURA_BUILDING)

function(pixaura_strict target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
        if(PIXAURA_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()
pixaura_strict(pixaura_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(core_test packages/core/tests/core_test.cpp)
    add_executable(c_consumer packages/core/tests/c_consumer.c)
    foreach(target core_test c_consumer)
        target_link_libraries(${target} PRIVATE pixaura_core)
        pixaura_strict(${target})
        add_test(NAME ${target} COMMAND ${target})
    endforeach()
    add_executable(document_test packages/core/tests/document_test.cpp packages/core/src/document.cpp)
    target_link_libraries(document_test PRIVATE pixaura_core)
    pixaura_strict(document_test)
    add_test(NAME document_test COMMAND document_test "${CMAKE_SOURCE_DIR}/tests/fixtures/image-document-v1.json")
    add_executable(document_c_consumer packages/core/tests/document_c_consumer.c)
    target_link_libraries(document_c_consumer PRIVATE pixaura_core)
    pixaura_strict(document_c_consumer)
    add_test(NAME document_c_consumer COMMAND document_c_consumer "${CMAKE_SOURCE_DIR}/tests/fixtures/image-document-v1.json" "${CMAKE_BINARY_DIR}/document-canonical.json")
    # Fault injection lives only in this independent test executable, never the DLL.
    add_executable(document_allocation_test packages/core/tests/document_allocation_test.cpp
        packages/core/src/document.cpp packages/core/src/document_api.cpp)
    target_include_directories(document_allocation_test PRIVATE packages/core/include)
    target_compile_features(document_allocation_test PRIVATE cxx_std_17)
    pixaura_strict(document_allocation_test)
    add_test(NAME document_allocation_test COMMAND document_allocation_test "${CMAKE_SOURCE_DIR}/tests/fixtures/image-document-v1.json")
    if(PIXAURA_SANITIZERS)
        set_tests_properties(core_test c_consumer document_test document_c_consumer document_allocation_test PROPERTIES ENVIRONMENT
            "ASAN_OPTIONS=detect_leaks=1:halt_on_error=1;UBSAN_OPTIONS=print_stacktrace=1:halt_on_error=1")
        add_executable(asan_negative packages/core/tests/asan_negative.cpp)
        target_link_libraries(asan_negative PRIVATE pixaura_core)
        add_executable(ubsan_negative packages/core/tests/ubsan_negative.cpp)
        foreach(kind asan ubsan)
            pixaura_strict(${kind}_negative)
            add_test(NAME ${kind}_instrumentation_active COMMAND ${CMAKE_COMMAND}
                "-DPIXAURA_PROBE=$<TARGET_FILE:${kind}_negative>"
                "-DPIXAURA_KIND=${kind}"
                "-DPIXAURA_LOG_DIR=${CMAKE_BINARY_DIR}/test-logs"
                -P "${CMAKE_SOURCE_DIR}/scripts/verify-sanitizer.cmake")
        endforeach()
    endif()
endif()

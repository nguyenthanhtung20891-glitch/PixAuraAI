function(pixaura_strict target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -fno-fast-math -ffp-contract=off)
        if(PIXAURA_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        endif()
    endif()
endfunction()

# Fault-injection targets compile the actual raster owner independently.
get_property(pixaura_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
foreach(pixaura_target IN LISTS pixaura_targets)
    get_target_property(pixaura_sources ${pixaura_target} SOURCES)
    if("packages/core/src/decode_api.cpp" IN_LIST pixaura_sources)
        foreach(policy_source capability budget dynamic_policy)
            if(NOT "packages/core/src/${policy_source}.cpp" IN_LIST pixaura_sources)
                target_sources(${pixaura_target} PRIVATE packages/core/src/${policy_source}.cpp)
            endif()
        endforeach()
    endif()
endforeach()
